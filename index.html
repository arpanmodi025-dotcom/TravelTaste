function syncTravelTasteUrlState({ replace = true, placeId, } = {}) { try { const url = new URL(window.location.href); const searchParams = url.searchParams; const setOrDelete = (key, value, defaultValue = '') => { const hasValue = value !== undefined && value !== null && String(value) !== '' && String(value) !== String(defaultValue); if (hasValue) { searchParams.set(key, String(value)); } else { searchParams.delete(key); } }; setOrDelete('city', selectedCity?.id); setOrDelete('q', globalSearchInput?.value?.trim()); setOrDelete('category', selectedCategory, 'all'); setOrDelete('style', selectedStyle, 'all'); setOrDelete('budget', selectedBudget, 'all'); setOrDelete('basis', selectedBudgetBasis, 'person'); setOrDelete('groupSize', selectedGroupSize, 4); setOrDelete('duration', selectedTripDuration, 'any'); if (placeId !== undefined) { setOrDelete('place', placeId); } const historyMethod = replace ? 'replaceState' : 'pushState'; history[historyMethod]( { ...(history.state || {}), travelTaste: true }, '', url ); } catch (error) { console.warn('Could not save Explore state to URL:', error); } } function restoreTravelTasteControlsFromUrl() { const params = new URL(window.location.href).searchParams; const searchQuery = params.get('q') || ''; selectedCategory = params.get('category') || 'all'; selectedStyle = params.get('style') || 'all'; selectedBudget = params.get('budget') || 'all'; selectedBudgetBasis = params.get('basis') || 'person'; selectedGroupSize = Number(params.get('groupSize')) || 4; selectedTripDuration = params.get('duration') || 'any'; if (globalSearchInput) { globalSearchInput.value = searchQuery; } if (clearGlobalSearch) { clearGlobalSearch.hidden = !searchQuery.trim(); } const controlValues = [ ['budgetBasis', selectedBudgetBasis], ['budgetGroupSize', String(selectedGroupSize)], ['tripDuration', selectedTripDuration], ]; for (const [id, value] of controlValues) { const control = $(id); if (control) control.value = value; } updateButtonStates('.category', 'category', selectedCategory); updateButtonStates('.style-card', 'style', selectedStyle); updateButtonStates('.budget-chip', 'budget', selectedBudget); updateBudgetControls(); } function updateButtonStates(selector, dataKey, selectedValue) { document.querySelectorAll(selector).forEach(button => { const isSelected = button.dataset[dataKey] === selectedValue; button.classList.toggle('active', isSelected); button.setAttribute('aria-pressed', String(isSelected)); }); }
